cmake_minimum_required(VERSION 3.10)

project(rknn_ppocr_system_parallel_worker)

set(
    MODEL_ZOO_ROOT
    /userdata/aidemo/rknn_model_zoo_ppocr_min
    CACHE PATH
    "RKNN Model Zoo source root"
)
add_subdirectory(${MODEL_ZOO_ROOT}/3rdparty 3rdparty.out)
add_subdirectory(${MODEL_ZOO_ROOT}/utils utils.out)

set(OpenCV_DIR ${MODEL_ZOO_ROOT}/3rdparty/opencv/opencv-linux-aarch64/share/OpenCV)
find_package(OpenCV REQUIRED)

add_executable(${PROJECT_NAME}
    ppocr_rknn_worker.cc
    postprocess.cc
    clipper.cc
    ppocr_system.cc
)

target_link_libraries(${PROJECT_NAME}
    fileutils
    imageutils
    imagedrawing
    ${OpenCV_LIBS}
    ${LIBRKNNRT}
)

set(THREADS_PREFER_PTHREAD_FLAG ON)
find_package(Threads REQUIRED)
target_link_libraries(${PROJECT_NAME} Threads::Threads)

target_include_directories(${PROJECT_NAME} PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}
    ${LIBRKNNRT_INCLUDES}
)
